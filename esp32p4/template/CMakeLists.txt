cmake_minimum_required(VERSION 3.16)
# PlatformIO's pre-build script generates these using its own Python interpreter.
# For direct IDF builds, run tools/prepare.py first.
include("${CMAKE_CURRENT_LIST_DIR}/build/paths.cmake")
set(KW_UI_DEV OFF)
set(KW_RENDER_BACKBUFFER ON)
set(KW_RENDER_PROFILE OFF)
set(KW_RENDER_BASELINE OFF)
set(KW_RENDER_PPA OFF)
set(KW_RENDER_INLINE ON)
set(KW_RENDER_INTERNAL OFF)
set(KW_RENDER_MODE DIRECT)
set(KW_RENDER_LINES 24)
set(KW_RENDER_THREAD_PRIO 3)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/src" "${KW_ROOT}/firmware/p4/components/lvgl")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(sloop_p4)
idf_component_get_property(UAC_DIR espressif__usb_host_uac COMPONENT_DIR)
set(PATCHED_UAC "${CMAKE_BINARY_DIR}/uac_host_p4.c")
execute_process(COMMAND "${PYTHON}" "${KW_ROOT}/firmware/p4/tools/patch_uac.py"
    "${UAC_DIR}/uac_host.c" "${PATCHED_UAC}" RESULT_VARIABLE PATCH_RESULT)
if(NOT PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "Pinned UAC correction failed")
endif()
get_target_property(UAC_SOURCES __idf_espressif__usb_host_uac SOURCES)
list(FILTER UAC_SOURCES EXCLUDE REGEX "(^|/)uac_host[.]c$")
list(APPEND UAC_SOURCES "${PATCHED_UAC}")
set_property(TARGET __idf_espressif__usb_host_uac PROPERTY SOURCES "${UAC_SOURCES}")
target_include_directories(__idf_espressif__usb_host_uac PRIVATE "${KW_ROOT}/firmware/p4/src")
